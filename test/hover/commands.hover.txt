export const banCommand = ban

    readonly options: {
        readonly days?: number;
        readonly log?: AnnouncementChannel<never> | TextChannel<never>;
        readonly reason?: "abuse" | "spam";

    };
}>

type BanOptions = {
    readonly days?: number;
    readonly log?: AnnouncementChannel<never> | TextChannel<never>;
    readonly reason?: "abuse" | "spam";
    readonly user: ResolvedUser;
}
